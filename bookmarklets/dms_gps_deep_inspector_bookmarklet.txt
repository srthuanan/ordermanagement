        var payload = {
            itv_requestObject: JSON.stringify({
                data: [{ vinCode: vin }],
                isUpdate: isUpdate
            })
        };

        try {
            var res = await fetch("/api/data/v9.2/itv_trackingvehicleposition", {
                method: "POST",
                headers: {
                    "OData-MaxVersion": "4.0",
                    "OData-Version": "4.0",
                    "Accept": "application/json",
                    "Content-Type": "application/json; charset=utf-8"
                },
                body: JSON.stringify(payload)
            });

            var duration = Math.round(performance.now() - startTime);
            var statusText = "HTTP " + res.status + " (" + duration + "ms)";
            document.getElementById("resStatus").textContent = statusText;

            var resJson = await res.json();
            var innerObj = null;
            try {
                innerObj = JSON.parse(resJson.itv_responseObject);
            } catch (e) {
                innerObj = resJson.itv_responseObject;
            }

            lastTestedData = {
                vin: vin,
                isUpdate: isUpdate,
                httpStatus: res.status,
                latencyMs: duration,
                response: resJson,
                parsedResponseObject: innerObj
            };

            // Analyze extracted data
            var packet = innerObj;
            if (innerObj && innerObj.data && Array.isArray(innerObj.data)) packet = innerObj.data[0];
            else if (Array.isArray(innerObj)) packet = innerObj[0];

            if (packet && typeof packet === "object") {
                var lat = packet.Lat || packet.lat || packet.latitude || packet.Latitude || "--";
                var lng = packet.Long || packet.long || packet.Lng || packet.lng || packet.longitude || packet.Longitude || "--";
                document.getElementById("resCoord").textContent = lat + ", " + lng;

                var tVal = packet.gpsTime || packet.deviceTime || packet.trackingTime || packet.time || packet.timestamp || packet.dateTime || packet.updatedTime || packet.createdDate || "Không có";
                document.getElementById("resTime").textContent = String(tVal);

                var spd = packet.Speed || packet.speed || packet.Velocity || packet.velocity || 0;
                var head = packet.Heading || packet.heading || packet.Bearing || packet.bearing || "--";
                document.getElementById("resSpeed").textContent = spd + " km/h | " + head + "°";

                if (lat !== "--" && lng !== "--") {
                    var gMapBtn = document.getElementById("btnOpenGoogleMaps");

                        window.open("https://www.google.com/maps?q=" + lat + "," + lng, "_blank");
                    };
                }
            } else {
                document.getElementById("resCoord").textContent = "Không tìm thấy";
                document.getElementById("resTime").textContent = "Không tìm thấy";

        } catch (e) {
            report.push("❌ Lỗi đọc itv_vinfastposition: " + e.message);
        }

        // 2. Check attributes of itv_vinfastposition
        try {
            var aRes = await fetch("/api/data/v9.2/EntityDefinitions(LogicalName='itv_vinfastposition')/Attributes?$select=LogicalName,DisplayName,AttributeType");
            var aJson = await aRes.json();
            var attrs = (aJson.value || []).map(function (a) {
                var disp = (a.DisplayName && a.DisplayName.UserLocalizedLabel) ? a.DisplayName.UserLocalizedLabel.Label : "";
                return a.LogicalName + (disp ? " (" + disp + ")" : "");
            });
            report.push("\n=== 2. CÁC CỘT CỦA BẢNG itv_vinfastposition (" + attrs.length + " cột) ===\n" + attrs.join(", "));
        } catch (e) {
            report.push("\n❌ Lỗi đọc thuộc tính itv_vinfastposition: " + e.message);
        }

        // 3. Scan WebResources for Tracking JavaScript code
        try {
            var wrRes = await _Xrm.WebApi.retrieveMultipleRecords("webresource", "?$select=name,displayname&$filter=contains(name, 'track') or contains(name, 'position') or contains(name, 'location') or contains(name, 'gps')&$top=25");
            report.push("\n=== 3. CÁC WEBRESOURCE JAVASCRIPT LIÊN QUAN ĐẾN ĐỊNH VỊ (" + wrRes.entities.length + " files) ===");
            wrRes.entities.forEach(function (w, i) {
                report.push((i + 1) + ". " + w.name + " (" + (w.displayname || "N/A") + ")");
            });
        } catch (e) {
            report.push("\n❌ Lỗi quét WebResources: " + e.message);
        }

        // 4. Check device tables
        try {
            var devRes = await _Xrm.WebApi.retrieveMultipleRecords("itv_deviceinfomation", "?$top=5");
            report.push("\n=== 4. BẢNG THIẾT BỊ itv_deviceinfomation (" + devRes.entities.length + " bản ghi) ===");
            if (devRes.entities.length > 0) {
                report.push(JSON.stringify(devRes.entities, null, 2));
            }
        } catch (e) {}

        el.textContent = report.join("\n");
    };
