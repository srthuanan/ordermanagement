    function isValidGuid(id) {
        return /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test((id || "").trim());

    // 1. Thu thập danh sách PO từ màn hình (DOM / Grid / Prompt)
    function detectSelectedPOsOnScreen() {
        var detected = [];
        var seenKeys = {};

        function addPoItem(id, num) {
            var cleanId = isValidGuid(id) ? id.trim() : "";
            var cleanNum = (num && typeof num === "string" && num.trim().length > 3) ? num.trim() : "";

            if (!cleanId && !cleanNum) return;

            var key = cleanId ? ("ID:" + cleanId.toLowerCase()) : ("NUM:" + cleanNum.toUpperCase());
            if (seenKeys[key]) return;
            seenKeys[key] = true;

            detected.push({ id: cleanId, number: cleanNum });
        }

        // A. Lấy từ Xrm Grid API
        try {
            var controls = _Xrm.Page ? _Xrm.Page.getControl() : null;
            if (controls && controls.forEach) {
                controls.forEach(function(c) {
                    if (c.getGrid && c.getGrid().getSelectedRows) {
                        var rows = c.getGrid().getSelectedRows();
                        if (rows && rows.forEach) {
                            rows.forEach(function(r) {
                                var data = r.getData ? r.getData() : null;
                                var entity = data ? data.getEntity() : null;
                                if (entity) {
                                    var id = (entity.getId() || "").replace(/[{}]/g, "");
                                    var num = entity.getPrimaryAttributeValue() || "";
                                    addPoItem(id, num);
                                }
                            });
                        }
                    }
                });
            }
        } catch(e){}

        // B. Quét dòng được tick checkbox trên giao diện DOM (UCI Grid / PowerApps Grid)
        try {
            var selRows = document.querySelectorAll('tr[aria-selected="true"], div[aria-selected="true"], [role="row"][aria-selected="true"], .ag-row-selected');
            selRows.forEach(function(el) {
                var isChecked = el.getAttribute('aria-selected') === 'true' || el.classList.contains('ag-row-selected') || el.querySelector('input[type="checkbox"]:checked, [role="checkbox"][aria-checked="true"]');
                if (isChecked) {
                    var rawId = el.getAttribute('data-id') || el.getAttribute('aria-rowid') || el.getAttribute('data-lp-id') || "";
                    var cleanId = rawId.replace(/[{}]/g, "").split(";")[0];
                    var text = el.innerText || el.textContent || "";
                    var match = text.match(/N\d{4,5}-PO-[\d\-]+/i);
                    var poNum = match ? match[0] : "";

                    addPoItem(cleanId, poNum);
                }
            });
        } catch(e){}

        // C. Nếu không tìm thấy bằng aria-selected, thử tìm tất cả checkbox đang checked trong bảng
        if (detected.length === 0) {
            try {
                var checkedInputs = document.querySelectorAll('input[type="checkbox"]:checked, [role="checkbox"][aria-checked="true"]');
                checkedInputs.forEach(function(chk) {
                    var row = chk.closest('tr, [role="row"], div[data-id]');
                    if (row) {
                        var rawId = row.getAttribute('data-id') || row.getAttribute('aria-rowid') || "";
                        var cleanId = rawId.replace(/[{}]/g, "").split(";")[0];
                        var text = row.innerText || row.textContent || "";
                        var match = text.match(/N\d{4,5}-PO-[\d\-]+/i);
                        var poNum = match ? match[0] : "";
                        addPoItem(cleanId, poNum);
                    }
                });
            } catch(e){}
        }

                    var formNum = formAttr ? formAttr.getValue() : "";
                    addPoItem(formId, formNum);
                }
            }
        } catch(e){}

        return detected;

            progressDiv.innerHTML = '🔍 Đang tra cứu ' + nums.length + ' số PO vừa nhập...';
            document.body.appendChild(progressDiv);

            loadPoDetailsAndShowModal(itemsToFetch);
        };

        // Thực thi gửi hàng loạt
        document.getElementById('btnStartBatchSend').onclick = function() {
            var selectedItems = poList.filter(function(p){ return p.selected && p.resultState !== 'success'; });
            if (selectedItems.length === 0) {
                alert("Vui lòng chọn ít nhất 1 Đơn Mua Hàng (PO) cần chuyển trạng thái Gửi!");
                return;
            }

            if (!confirm("⚠️ Bạn có chắc chắn muốn chuyển trạng thái GỬI (SUBMIT) cho " + selectedItems.length + " Đơn Mua Hàng đã chọn không?")) {
                return;
            }

            var btn = document.getElementById('btnStartBatchSend');
            btn.disabled = true;
            btn.style.opacity = '0.6';
            btn.innerHTML = '⚙️ Đang thực thi gửi...';
            progressContainer.style.display = 'flex';

            var successCount = 0;
            var errorCount = 0;

            function processIndex(i) {
                if (i >= selectedItems.length) {
                    btn.disabled = false;
                    btn.style.opacity = '1';
                    btn.innerHTML = '✅ HOÀN TẤT GỬI';
                    document.getElementById('poProgressStatusText').innerHTML = '🎉 Hoàn tất! Thành công: <b>' + successCount + '</b> | Thất bại: <b>' + errorCount + '</b>';
                    document.getElementById('poProgressBarFill').style.width = '100%';
                    document.getElementById('poProgressPercentText').textContent = '100%';

                    alert("🎉 Đã hoàn tất chuyển trạng thái Gửi cho các Đơn Mua Hàng!\n• Thành công: " + successCount + "\n• Thất bại: " + errorCount);

                    try {
                        if (_Xrm.Page && _Xrm.Page.data) _Xrm.Page.data.refresh(true);
                        else location.reload();
                    } catch(e) { location.reload(); }
                    return;
                }

                var current = selectedItems[i];
                current.resultState = 'running';
                var percent = Math.round((i / selectedItems.length) * 100);
                document.getElementById('poProgressBarFill').style.width = percent + '%';
                document.getElementById('poProgressPercentText').textContent = percent + '%';
                document.getElementById('poProgressStatusText').innerHTML = '⚙️ Đang gửi đơn (' + (i + 1) + '/' + selectedItems.length + '): <b>' + current.number + '</b>...';

                var stCell = document.getElementById('po-status-' + current.id);
                if (stCell) stCell.innerHTML = '<span style="color:#38bdf8;background:#38bdf820;padding:2px 8px;border-radius:4px;font-size:11px;font-weight:bold;">⚙️ Đang gửi...</span>';

                // Payload cập nhật trạng thái Gửi (Submit / Send PO in VinFast DMS)
                var updatePayload = {
                    "xts_handling": 2
                };

                _Xrm.WebApi.updateRecord("xts_purchaseorder", current.id, updatePayload).then(function() {
                    current.resultState = 'success';
                    current.resultMsg = 'Đã chuyển trạng thái Gửi thành công';
                    successCount++;
                    if (stCell) stCell.innerHTML = '<span style="color:#10b981;background:#10b98120;padding:2px 8px;border-radius:4px;font-size:11px;font-weight:bold;border:1px solid #10b98140;">✅ ĐÃ GỬI (SENT)</span>';
                    setTimeout(function() { processIndex(i + 1); }, 600);
                }, function(err) {
                    _Xrm.WebApi.updateRecord("xts_purchaseorder", current.id, { "xts_status": 2 }).then(function() {
                        current.resultState = 'success';
                        current.resultMsg = 'Đã cập nhật trạng thái Gửi (xts_status)';
                        successCount++;
                        if (stCell) stCell.innerHTML = '<span style="color:#10b981;background:#10b98120;padding:2px 8px;border-radius:4px;font-size:11px;font-weight:bold;border:1px solid #10b98140;">✅ ĐÃ GỬI (SENT)</span>';
                        setTimeout(function() { processIndex(i + 1); }, 600);
                    }, function(err2) {
                        current.resultState = 'error';
                        current.resultMsg = err.message || err2.message || JSON.stringify(err);
                        errorCount++;
                        if (stCell) stCell.innerHTML = '<span style="color:#ef4444;background:#ef444420;padding:2px 8px;border-radius:4px;font-size:11px;font-weight:bold;border:1px solid #ef444440;" title="' + current.resultMsg.replace(/"/g, '&quot;') + '">❌ Lỗi gửi</span>';
                        setTimeout(function() { processIndex(i + 1); }, 600);
                    });
                });
            }

            processIndex(0);
        };
    }

    // Tiến hành tải chi tiết
    if (detectedItems.length > 0) {
        loadPoDetailsAndShowModal(detectedItems);
    } else {
        loadPoDetailsAndShowModal(null);
    }
})();
