                    // Bước A: Tạo Báo giá trước (nếu bật checkbox)
                    if (shouldCreateQuote) {
                        updateLog("📑 Đang tự động tạo Báo giá xe mới (Quote) cho khách hàng " + targetCust.name + "...");
                        
                        var quotePayload = {
                            "xts_potentialcustomerid@odata.bind": "/accounts(" + custIdClean + ")",
                            "xts_customernumber": targetCust.accountnumber || "",
                            "xts_transactiondate": new Date().toISOString(),
                            "xts_status": 1, // Mở
                            "statecode": 0
                        };

                        function bindQuoteLookup(field, entitySet) {
                            var val = fullTpl["_" + field + "_value"];
                            if (val) quotePayload[field + "@odata.bind"] = "/" + entitySet + "(" + val + ")";
                        }

                        bindQuoteLookup("xts_salespersonid", "xts_employees");
                        bindQuoteLookup("xts_productid", "xts_products");
                        bindQuoteLookup("xts_productconfigurationid", "xts_productconfigurations");
                        bindQuoteLookup("xts_productexteriorcolorid", "xts_productexteriorcolors");
                        bindQuoteLookup("xts_productinteriorcolorid", "xts_productinteriorcolors");
                        bindQuoteLookup("xts_siteid", "xts_sites");
                        bindQuoteLookup("xts_businessunitid", "businessunits");
                        bindQuoteLookup("xts_ordertypeid", "xts_ordertypes");
                        bindQuoteLookup("xts_termsofpaymentid", "xts_termofpayments");
                        bindQuoteLookup("xts_vehiclepricelistid", "xts_vehicleprices");
                        bindQuoteLookup("xts_consumptiontax1id", "xts_consumptiontaxes");
                        bindQuoteLookup("transactioncurrencyid", "transactioncurrencies");

                        if (fullTpl.xts_vehicleamount) quotePayload["xts_vehicleamount"] = fullTpl.xts_vehicleamount;
                        if (fullTpl.xts_netamountaftertax) quotePayload["xts_netamountaftertax"] = fullTpl.xts_netamountaftertax;
                        if (fullTpl.xvf_grandtotal) quotePayload["xvf_grandtotal"] = fullTpl.xvf_grandtotal;
                        if (fullTpl.xts_productdescription) quotePayload["xts_productdescription"] = fullTpl.xts_productdescription;

                        _Xrm.WebApi.createRecord("xts_newvehiclesalesquote", quotePayload).then(function(newQuote){
                            var newQuoteId = (newQuote.id || "").replace(/[{}]/g, "");
                            _Xrm.WebApi.retrieveRecord("xts_newvehiclesalesquote", newQuoteId, "?$select=xts_newvehiclesalesquotenumber").then(function(qRec){
                                var qNum = qRec.xts_newvehiclesalesquotenumber || "Báo giá mới";
                                updateLog("Đã tạo Báo giá " + qNum + ". Tiếp tục tạo VSO...");
                                proceedCreateVSO(newQuoteId, qNum);
                            }).catch(function(){
                                proceedCreateVSO(newQuoteId, "Báo giá mới");
                            });
                        }).catch(function(errQuote){
                            console.warn("Không thể tạo Quote, tiếp tục tạo VSO trực tiếp:", errQuote);
                            updateLog("Không tạo được Quote tự động, đang tạo VSO trực tiếp...");
                            proceedCreateVSO(null, null);
                        });

                    } else {
                        proceedCreateVSO(null, null);
                    }

                }).catch(function(errRead){
                    btnExec.disabled = false;
                    btnExec.style.opacity = "1";
                    updateLog("Lỗi đọc dữ liệu đơn mẫu: " + (errRead.message || errRead), true);
                    alert("❌ Lỗi đọc đơn mẫu: " + (errRead.message || errRead));
                });
            };
        }

        // Nút Đóng / Hủy
        var btnClose = modal.querySelector("#btn-close-creator");
        var btnCancel = modal.querySelector("#btn-cancel-create");
        if (btnClose) btnClose.onclick = function(){ overlay.remove(); };
        if (btnCancel) btnCancel.onclick = function(){ overlay.remove(); };

        loadRecentVSOs();
        searchCustomers("");

    } catch(err) {
        alert("❌ Lỗi khởi tạo Trợ lý Tạo Đơn VSO: " + (err.message || err));
        console.error("VSO Creator Error:", err);
    }
})();
