    if (!x) {
        for (var i = 0; i < frames.length; i++) {
            try { if (frames[i].Xrm) { x = frames[i].Xrm; break; } } catch(e){}
        }
    }

    var successId = "d3e128ed-859f-f111-b8dc-70a8a5045a79"; // N31913-VDO-26-08-0149

        var report = [];

        report.push(`• xts_status: ${recSuccess["xts_status@OData.Community.Display.V1.FormattedValue"] || recSuccess.xts_status}`);
        report.push(`• statuscode: ${recSuccess["statuscode@OData.Community.Display.V1.FormattedValue"] || recSuccess.statuscode} (${recSuccess.statuscode})`);
        report.push(`• statecode: ${recSuccess["statecode@OData.Community.Display.V1.FormattedValue"] || recSuccess.statecode} (${recSuccess.statecode})`);

            report.push(`• VSO itv_externalstatus: ${vsoS["itv_externalstatus@OData.Community.Display.V1.FormattedValue"] || vsoS.itv_externalstatus}`);
            report.push(`• VSO itv_codeerror: ${vsoS.itv_codeerror || "Không có"}`);

        // Sao chép thuộc tính từ phiếu thành công sang phiếu 0150 nếu đang mở

                try {
                    await x.WebApi.updateRecord("xts_newvehicledeliveryorder", currentId, copyPayload);
