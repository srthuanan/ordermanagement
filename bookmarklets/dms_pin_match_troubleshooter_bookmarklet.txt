                    badge: "🔴 Trạng Thái PIN Chưa Sẵn Sàng",

                    desc: `Bản ghi <code>xts_inventserial</code> (ID: <code>${bSerial.xts_inventserialid}</code>) đang có trạng thái <b>${bStatus || bSerialStatus}</b>. Để ghép được xe, sê-ri PIN bắt buộc phải ở trạng thái <b>Sẵn sàng (Available / Ready)</b>.`

    window.dmsTriggerReleasePr = async function(prId, prNumber) {
        if (!confirm(`⚠️ Bạn có chắc chắn muốn PHÁT HÀNH Phiếu Nhập Kho [${prNumber || prId}] không?`)) return;
        var footEl = document.getElementById("dmsStatusFoot");
        footEl.innerHTML = `⏳ Đang phát hành Phiếu Nhập Kho ${prNumber}...`;
        try {
            await _Xrm.WebApi.updateRecord("xts_purchasereceipt", prId, { "xts_handling": 2, "xts_calculatetax": true });
            alert(`🎉 ĐÃ PHÁT HÀNH THÀNH CÔNG Phiếu Nhập Kho ${prNumber}!\n\nSê-ri PIN đã được chuyển sang trạng thái Sẵn Sàng.`);
            runDiagnostics(diagnosticState.keySearch);
        } catch(err) {
            alert(`❌ Lỗi: ` + (err.message || JSON.stringify(err)));
            footEl.innerHTML = `❌ Lỗi: ` + (err.message || String(err));
        }
    };
