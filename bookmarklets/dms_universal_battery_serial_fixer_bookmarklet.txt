    async function analyzeVehicle(key) {
        key = (key || "").trim();
        if (!key) {
            alert("⚠️ Vui lòng nhập Số Tồn Kho, Số Khung (VIN) hoặc Mã PIN!");
            return;

        footEl.innerHTML = `🔍 Đang quét xe: [${key}]...`;
        bodyEl.innerHTML = `<div style="text-align:center;padding:40px 0;color:#38bdf8;">⚡ Đang truy vấn OData CRM cho xe [${key}]...</div>`;

        currentData = { vehicle: null, pinSerialRecord: null, allSerials: [], pr: null };

        try {
            // 1. Tìm Xe trong xts_inventorynewvehicle
            var vehFilter = "";
            if (/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(key)) {
                vehFilter = `xts_inventorynewvehicleid eq ${key}`;
            } else if (key.length === 17) {
                vehFilter = `xts_chassisnumber eq '${key}'`;
            } else if (key.indexOf("-SN-") !== -1 || key.indexOf("SN") !== -1) {
                vehFilter = `xts_stocknumber eq '${key}'`;
            } else {
                vehFilter = `xts_stocknumber eq '${key}' or xts_chassisnumber eq '${key}'`;
            }

            var vehRes = await _Xrm.WebApi.retrieveMultipleRecords("xts_inventorynewvehicle", `?$filter=${encodeURIComponent(vehFilter)}&$top=1`);
            if (vehRes && vehRes.entities && vehRes.entities.length > 0) {
                currentData.vehicle = vehRes.entities[0];
            } else {
                bodyEl.innerHTML = `<div style="color:#f87171;text-align:center;padding:30px;">❌ Không tìm thấy bản ghi xe nào với từ khóa: <b>${key}</b></div>`;
                footEl.innerHTML = "❌ Không tìm thấy xe";
                return;
            }

            var veh = currentData.vehicle;
            var pinCode = veh.itv_batteryserial;
            var poNum = veh.xts_referencenumber;

            // 2. Tìm Sê-ri PIN trong xts_inventserial
            if (pinCode) {
                try {
                    var sRes = await _Xrm.WebApi.retrieveMultipleRecords("xts_inventserial", `?$filter=xts_serialnumber eq '${pinCode}'&$top=5`);
                    if (sRes && sRes.entities && sRes.entities.length > 0) {
                        currentData.pinSerialRecord = sRes.entities[0];
                        currentData.allSerials = sRes.entities;
                    }
                } catch(ePin){
                    console.warn("Lỗi sê-ri PIN:", ePin);
                }
            }

            // 3. Tìm PR (Chỉ dùng xts_purchasereceiptnumber)
            if (poNum) {
                try {
                    var prRes = await _Xrm.WebApi.retrieveMultipleRecords("xts_purchasereceipt", `?$filter=contains(xts_purchasereceiptnumber, '${poNum}')&$top=1`);
                    if (prRes && prRes.entities && prRes.entities.length > 0) {
                        currentData.pr = prRes.entities[0];
                    }
                } catch(ePr){
                    console.warn("Lỗi PR:", ePr);
                }
            }

            footEl.innerHTML = `✅ Đã quét xong xe [${veh.xts_stocknumber}] lúc ${(new Date()).toLocaleTimeString("vi-VN")}`;
            renderUniTab("u-tab-diag");

        } catch(err) {
            console.error("Lỗi:", err);
            bodyEl.innerHTML = `<div style="color:#f87171;padding:20px;">❌ Lỗi truy vấn: ${err.message || JSON.stringify(err)}</div>`;
            footEl.innerHTML = "❌ Lỗi truy vấn";
        }
    }

    window.dmsExecuteAutoFix = async function() {
        var veh = currentData.vehicle;
        if (!veh) {
            alert("⚠️ Vui lòng quét thông tin xe trước!");
            return;
        }

        var pinCode = veh.itv_batteryserial;
        var vin = veh.xts_chassisnumber;
        var poNum = veh.xts_referencenumber;
        var vehId = veh.xts_inventorynewvehicleid;

        if (!pinCode) {
            alert("⚠️ Xe này chưa có mã Sê-ri PIN trên trường itv_batteryserial!");
            return;
        }

        footEl.innerHTML = "⏳ Đang kích hoạt Sẵn Sàng Sê-ri PIN...";
        try {
            var targetPinSerialId = currentData.pinSerialRecord ? currentData.pinSerialRecord.xts_inventserialid : "";

            if (targetPinSerialId) {
                await _Xrm.WebApi.updateRecord("xts_inventserial", targetPinSerialId, {
                    "xts_availabilitystatus": 6,
                    "xts_stockstatus": 3
                });
            } else {
                var createPayload = {
                    "xts_serialnumber": pinCode,
                    "xts_description": (poNum || "") + " - PIN theo xe " + vin,
                    "xts_availabilitystatus": 6,
                    "xts_stockstatus": 3,
                    "xts_vehiclesource": 1,
                    "statecode": 0,
                    "statuscode": 1
                };
                var newRes = await _Xrm.WebApi.createRecord("xts_inventserial", createPayload);
                targetPinSerialId = newRes.id;
            }

            try {
                await _Xrm.WebApi.updateRecord("xts_inventorynewvehicle", vehId, {
                    "itv_serialid@odata.bind": `/xts_inventserials(${targetPinSerialId})`,
                    "itv_batteryserial": pinCode
                });
            } catch(eBind){
                console.warn("Bind vehicle:", eBind);
            }

            if (currentData.pr && (currentData.pr.xts_status === 1 || currentData.pr.xts_handling === 1)) {
                try {
                    await _Xrm.WebApi.updateRecord("xts_purchasereceipt", currentData.pr.xts_purchasereceiptid, {
                        "xts_handling": 2,
                        "xts_calculatetax": true
                    });
                } catch(ePr){}
            }

            alert("🎉 ĐÃ KÍCH HOẠT SẴN SÀNG SÊ-RI PIN CHO XE [" + veh.xts_stocknumber + "] THÀNH CÔNG!\n\nBây giờ xe đã đủ điều kiện để Ghép vào Đơn hàng.");

        } catch(errFix) {

            footEl.innerHTML = "❌ Lỗi kích hoạt: " + (errFix.message || String(errFix));
        }
    };
