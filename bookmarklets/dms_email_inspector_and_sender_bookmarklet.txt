    // Hàm build payload CRM
    function buildEmailPayload() {
        var fromId = document.getElementById('send-from-input').value.trim();
        var fromType = document.getElementById('send-from-type').value.trim(); // systemuser or queue
        var toEmail = document.getElementById('send-to-input').value.trim();
        var regardingType = document.getElementById('send-regarding-type').value.trim();
        var regardingId = document.getElementById('send-regarding-id').value.trim();
        var subject = document.getElementById('send-subject-input').value.trim();
        var bodyHtml = document.getElementById('send-body-input').value.trim();

        if (!toEmail) {
            alert("⚠️ Vui lòng nhập địa chỉ Email người nhận (To)!");
            throw new Error("Missing To");
        }

        var activityParties = [];
        if (fromId) {
            var partyObj = { "participationtypemask": 1 };
            if (fromType === 'queue') {
                partyObj["partyid_queue@odata.bind"] = "/queues(" + fromId + ")";
            } else {
                partyObj["partyid_systemuser@odata.bind"] = "/systemusers(" + fromId + ")";
            }
            activityParties.push(partyObj);
        }

        activityParties.push({
            "addressused": toEmail,
            "participationtypemask": 2
        });

        var payload = {
            "subject": subject,
            "description": bodyHtml,
            "directioncode": true,
            "email_activity_parties": activityParties
        };

        if (regardingType && regardingId) {
            payload["regardingobjectid_" + regardingType + "@odata.bind"] = "/" + regardingType + "s(" + regardingId + ")";
        }

        return payload;
    }

    // Nút 2: Gửi Native Web API
    document.getElementById('btn-execute-send').onclick = function() {
        try {
            var payload = buildEmailPayload();
            logMsg("🚀 TIẾN TRÌNH GỬI NATIVE WEB API...", "warn");

            var url = clientUrl + "/api/data/v9.2/emails";
            fetch(url, {
                method: "POST",
                headers: {
                    "OData-MaxVersion": "4.0",
                    "OData-Version": "4.0",
                    "Accept": "application/json",
                    "Content-Type": "application/json; charset=utf-8",
                    "Prefer": "return=representation"
                },
                body: JSON.stringify(payload)
            })
            .then(function(res) {
                if (!res.ok) return res.json().then(function(err){ throw err; });
                var entityIdHeader = res.headers.get("OData-EntityId");
                return res.json().catch(function(){ return {}; }).then(function(data) {
                    var emailId = data.activityid;
                    if (!emailId && entityIdHeader) {
                        var match = entityIdHeader.match(/\(([^)]+)\)/);
                        if (match) emailId = match[1];
                    }
                    return emailId;
                });
            })
            .then(function(emailId) {
                if (!emailId) throw new Error("Không trích xuất được Email GUID!");
                logMsg("✅ Đã tạo email ID: " + emailId, "ok");
                logMsg("Đang gọi Action SendEmail...");

                var sendUrl = clientUrl + "/api/data/v9.2/emails(" + emailId + ")/Microsoft.Dynamics.CRM.SendEmail";
                return fetch(sendUrl, {
                    method: "POST",
                    headers: {
                        "OData-MaxVersion": "4.0",
                        "OData-Version": "4.0",
                        "Accept": "application/json",
                        "Content-Type": "application/json; charset=utf-8"
                    },
                    body: JSON.stringify({ "IssueSend": true })
                }).then(function(sendRes) {
                    if (!sendRes.ok) return sendRes.json().then(function(e){ throw e; });
                    logMsg("🎉 Đã gửi lệnh phát qua CRM SendEmail!", "ok");
                    alert("🎉 Gửi lệnh SendEmail thành công!");
                    loadEmailHistory();
                });
            })
            .catch(function(err) {
                var errMsg = err.message || JSON.stringify(err);
                logMsg("❌ THẤT BẠI: " + errMsg, "err");
                alert("❌ Lỗi: " + errMsg);
            });
        } catch(e){}
    };

    var cachedEmails = [];

    function loadEmailHistory() {
        var container = document.getElementById('email-list-container');
        container.innerHTML = '<div style="text-align:center;padding:30px;color:#38bdf8;font-size:12px;">⏳ Đang tải danh sách email từ DMS...</div>';

        var query = 'emails?$top=35&$orderby=createdon desc&$select=activityid,subject,description,statuscode,statecode,createdon,modifiedon,_regardingobjectid_value,mimetype&$expand=email_activity_parties($select=activitypartyid,participationtypemask,addressused,_partyid_value)';
        
        apiFetch(query).then(function(res) {
            cachedEmails = res.value || [];
            renderEmailList(cachedEmails);
        }).catch(function(err) {
            apiFetch('emails?$top=35&$orderby=createdon desc&$select=activityid,subject,description,statuscode,statecode,createdon,modifiedon,_regardingobjectid_value').then(function(res2) {
                cachedEmails = res2.value || [];
                renderEmailList(cachedEmails);
            }).catch(function(err2) {
                container.innerHTML = '<div style="padding:20px;color:#ef4444;font-size:12px;">Lỗi tải lịch sử email: ' + (err2.message || JSON.stringify(err2)) + '</div>';
            });
        });

    function renderEmailList(list) {
        var container = document.getElementById('email-list-container');
        if (!list || list.length === 0) {
            container.innerHTML = '<div style="text-align:center;padding:30px;color:#64748b;font-size:12px;">Không có email nào trên hệ thống.</div>';
            return;
        }

        var html = '';
        list.forEach(function(em) {
            var subject = em.subject || "(Không có tiêu đề)";
            var dateStr = em.createdon ? new Date(em.createdon).toLocaleString('vi-VN') : "";
            
            var statusBadge = '';
            if (em.statuscode === 3) statusBadge = '<span style="background:rgba(34,197,94,0.15);color:#22c55e;padding:2px 6px;border-radius:4px;font-size:10px;font-weight:700;">Sent</span>';
            else if (em.statuscode === 1) statusBadge = '<span style="background:rgba(148,163,184,0.15);color:#94a3b8;padding:2px 6px;border-radius:4px;font-size:10px;font-weight:700;">Draft</span>';
            else if (em.statuscode === 6 || em.statuscode === 7) statusBadge = '<span style="background:rgba(245,158,11,0.15);color:#f59e0b;padding:2px 6px;border-radius:4px;font-size:10px;font-weight:700;">Pending</span>';
            else if (em.statuscode === 8) statusBadge = '<span style="background:rgba(239,68,68,0.15);color:#ef4444;padding:2px 6px;border-radius:4px;font-size:10px;font-weight:700;">Failed</span>';
            else statusBadge = '<span style="background:rgba(56,189,248,0.15);color:#38bdf8;padding:2px 6px;border-radius:4px;font-size:10px;font-weight:700;">Status ' + em.statuscode + '</span>';

            html += '<div class="dms-email-item" data-id="' + em.activityid + '" style="padding:12px;margin-bottom:6px;background:#0d1322;border:1px solid rgba(255,255,255,0.06);border-radius:8px;cursor:pointer;transition:all 0.15s;">' +
                '<div style="display:flex;justify-content:space-between;align-items:center;margin-bottom:4px;">' +
                    '<span style="font-size:11px;color:#94a3b8;">' + dateStr + '</span>' +
                    statusBadge +
                '</div>' +
                '<div style="font-weight:700;font-size:12.5px;color:#f8fafc;margin-bottom:4px;white-space:nowrap;overflow:hidden;text-overflow:ellipsis;">' + subject + '</div>' +
                '<div style="font-size:11px;color:#64748b;display:flex;gap:6px;align-items:center;">' +
                    '<span>ID: ' + em.activityid.substring(0,8) + '...</span>' +
                    (em._regardingobjectid_value ? '<span style="color:#a855f7;">• Regarding: ' + em._regardingobjectid_value.substring(0,8) + '</span>' : '') +
                '</div>' +
            '</div>';
        });

        container.innerHTML = html;

        container.querySelectorAll('.dms-email-item').forEach(function(item) {
            item.onclick = function() {
                container.querySelectorAll('.dms-email-item').forEach(function(i){ i.style.borderColor = 'rgba(255,255,255,0.06)'; i.style.background = '#0d1322'; });
                item.style.borderColor = '#38bdf8';
                item.style.background = '#1e293b';
                var emailId = item.getAttribute('data-id');
                var targetEmail = cachedEmails.find(function(e){ return e.activityid === emailId; });
                if (targetEmail) renderEmailDetail(targetEmail);
            };
        });
    }

    function renderEmailDetail(em) {

        var parties = em.email_activity_parties || [];

        var fromStr = fromParty.map(function(p){ return p.addressused || p._partyid_value || "User"; }).join(', ') || em.sender || "Chưa rõ";
        var toStr = toParties.map(function(p){ return p.addressused || p._partyid_value || "Customer"; }).join(', ') || em.to || "Chưa rõ";
