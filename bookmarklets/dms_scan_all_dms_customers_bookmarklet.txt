javascript:(function(){
    var x = typeof Xrm !== "undefined" ? Xrm : null;
    if (!x) {
        for (var i = 0; i < frames.length; i++) {
            try { if (frames[i].Xrm && frames[i].Xrm.WebApi) { x = frames[i].Xrm; break; } } catch (e) {}
        }
    }
    if (!x && window.top) {
        try { if (window.top.Xrm && window.top.Xrm.WebApi) x = window.top.Xrm; } catch (e) {}
    }

    if (!x || !x.WebApi) {
        alert("❌ Không tìm thấy CRM Xrm.WebApi! Vui lòng đảm bảo bạn đang mở trang VinFast DMS (Dynamics 365).");
        return;
    }

    var progressDiv = document.createElement('div');
    progressDiv.style = 'position:fixed;top:15px;right:15px;background:#0f172a;color:#ffffff;padding:16px 22px;z-index:9999999;border-radius:12px;font-family:system-ui,-apple-system,sans-serif;font-size:14px;box-shadow:0 20px 25px -5px rgba(0,0,0,0.5);border:1px solid #3b82f6;font-weight:600;display:flex;align-items:center;gap:12px;';
    progressDiv.innerHTML = '<span style="font-size:20px;">⚡</span> <span id="sync-status-text">Đang chuẩn bị quét TOÀN BỘ Mã Khách Hàng trên DMS...</span>';
    document.body.appendChild(progressDiv);

    function updateStatus(text) {
        var el = document.getElementById("sync-status-text");
        if (el) el.innerHTML = text;
    }

    var customerMap = new Map();

    function addOrUpdateCust(cNum, name, phone, oneId, vUserId, tier, vsoNum, source) {
        if (!cNum) return;
        cNum = String(cNum).trim();
        var existing = customerMap.get(cNum) || {
            ma_kh: cNum,
            ten_kh: name || "Chưa rõ",
            sdt: phone || "",
            one_id: oneId || "",
            vclub_user_id: vUserId || "",
            hang_vinclub: tier || "",
            cac_don_vso: [],
            co_vinclub: false
        };

        if (name && (existing.ten_kh === "Chưa rõ" || !existing.ten_kh)) existing.ten_kh = name;
        if (phone && !existing.sdt) existing.sdt = phone;
        if (oneId && !existing.one_id) existing.one_id = oneId;
        if (vUserId && !existing.vclub_user_id) existing.vclub_user_id = vUserId;
        if (tier && (!existing.hang_vinclub || existing.hang_vinclub === "Thành viên")) existing.hang_vinclub = tier;
        
        if (vsoNum && existing.cac_don_vso.indexOf(vsoNum) === -1) {
            existing.cac_don_vso.push(vsoNum);
        }

        if (existing.vclub_user_id || existing.hang_vinclub || existing.one_id || source === "Profile VinClub") {
            existing.co_vinclub = true;
        }

        customerMap.set(cNum, existing);
    }

    // Helper fetch with pagination
    function fetchAllRecords(entityName, query, accumulated) {
        accumulated = accumulated || [];
        return x.WebApi.retrieveMultipleRecords(entityName, query).then(function(res) {
            var items = accumulated.concat(res.entities || []);
            updateStatus('Đang nạp bảng <strong>' + entityName + '</strong>: ' + items.length + ' dòng...');
            var nextLink = res["@odata.nextLink"] || res.nextLink;
            if (nextLink) {
                var nextQuery = nextLink.substring(nextLink.indexOf('?'));
                return fetchAllRecords(entityName, nextQuery, items);
            }
            return items;
        });
    }

    // Step 1: Quét TOÀN BỘ bảng Customer Account (không filter bớt)
    updateStatus('Đang quét TOÀN BỘ Khách hàng trong bảng <strong>account</strong>...');
    var pAccount = fetchAllRecords("account", "?$select=accountnumber,name,telephone1,itv_vgcustomerid,_itv_customerexternalprofile_value&$filter=statecode eq 0")
    .then(function(accounts) {
        accounts.forEach(function(acc) {
            if (acc.accountnumber) {
                addOrUpdateCust(acc.accountnumber, acc.name, acc.telephone1, acc.itv_vgcustomerid, "", "", "", "Account");
            }
        });
    }).catch(function(e){ console.warn("Lỗi fetch account:", e); });

    // Step 2: Quét TOÀN BỘ bảng Hồ sơ VinClub (itv_customerexternalprofile)
    var pProfile = fetchAllRecords("itv_customerexternalprofile", "?$select=itv_vclubuserid,itv_vclubtiercode,itv_customernumber&$filter=statecode eq 0")
    .then(function(profiles) {
        profiles.forEach(function(p) {
            if (p.itv_customernumber) {
                addOrUpdateCust(p.itv_customernumber, "", "", "", p.itv_vclubuserid, p.itv_vclubtiercode, "", "Profile VinClub");
            }
        });
    }).catch(function(e){ console.warn("Lỗi fetch profiles:", e); });

    // Step 3: Quét TOÀN BỘ Đơn hàng Bán xe (xts_newvehiclesalesorder)
    var pOrder = fetchAllRecords("xts_newvehiclesalesorder", "?$select=xts_newvehiclesalesordernumber,xts_customernumber,itv_vinclubtiercode,itv_vinclubinterfacelog&$expand=xts_potentialcustomerid($select=accountnumber,name,itv_vgcustomerid,telephone1)&$filter=statecode eq 0")
    .then(function(orders) {
        orders.forEach(function(ord) {
            var cNum = ord.xts_customernumber || (ord.xts_potentialcustomerid ? ord.xts_potentialcustomerid.accountnumber : "");
            var cName = ord.xts_potentialcustomerid ? ord.xts_potentialcustomerid.name : "";
            var phone = ord.xts_potentialcustomerid ? ord.xts_potentialcustomerid.telephone1 : "";
            var oneId = ord.xts_potentialcustomerid ? ord.xts_potentialcustomerid.itv_vgcustomerid : "";
            var vsoNum = ord.xts_newvehiclesalesordernumber;
            var tier = ord.itv_vinclubtiercode || "";
            var vUserId = "";

            if (ord.itv_vinclubinterfacelog) {
                try {
                    var parsed = JSON.parse(ord.itv_vinclubinterfacelog);
                    if (parsed && parsed.data) {
                        if (parsed.data.vclub_user_id) vUserId = String(parsed.data.vclub_user_id);
                        if (parsed.data.tier && parsed.data.tier.tier_name) {
                            tier = parsed.data.tier.tier_name;
                        }
                    }
                } catch(e) {}
            }

            if (cNum) {
                addOrUpdateCust(cNum, cName, phone, oneId, vUserId, tier, vsoNum, "Đơn VSO");
            }
        });
    }).catch(function(e){ console.warn("Lỗi fetch orders:", e); });

    Promise.all([pAccount, pProfile, pOrder]).then(function() {
        progressDiv.remove();
        var allCustomers = Array.from(customerMap.values());
        showAllCustomersUI(allCustomers);
    }).catch(function(err) {
        progressDiv.remove();
        alert("❌ Lỗi khi quét toàn bộ danh sách Khách hàng: " + (err.message || err));
    });

    function showAllCustomersUI(list) {
        var old = document.getElementById("dms-all-cust-overlay");
        if (old) old.remove();

        var totalCount = list.length;
        var vcCount = list.filter(function(i){ return i.co_vinclub; }).length;

        var overlay = document.createElement("div");
        overlay.id = "dms-all-cust-overlay";
        overlay.style = "position:fixed;top:0;left:0;width:100%;height:100%;background:rgba(15,23,42,0.85);backdrop-filter:blur(10px);z-index:9999999;display:flex;align-items:center;justify-content:center;font-family:system-ui,-apple-system,sans-serif;";

        var modal = document.createElement("div");
        modal.style = "width:92%;max-width:1200px;max-height:88vh;background:#1e293b;border-radius:16px;border:1px solid #334155;box-shadow:0 25px 50px -12px rgba(0,0,0,0.5);display:flex;flex-direction:column;overflow:hidden;color:#f8fafc;";

        // Header
        var header = document.createElement("div");
        header.style = "padding:20px 24px;background:#0f172a;border-bottom:1px solid #334155;display:flex;justify-content:space-between;align-items:center;";
        header.innerHTML = '<div style="display:flex;align-items:center;gap:12px;">' +
                           '<span style="font-size:28px;">📇</span>' +
                           '<div><h2 style="margin:0;font-size:18px;font-weight:700;color:#38bdf8;">DANH SÁCH TOÀN BỘ MÃ KHÁCH HÀNG TỒN TẠI TRÊN DMS</h2>' +
                           '<p style="margin:4px 0 0 0;font-size:13px;color:#94a3b8;">Tổng cộng: <strong style="color:#ffffff;">' + totalCount + '</strong> Khách hàng | Có VinClub: <strong style="color:#10b981;">' + vcCount + '</strong> Khách hàng</p></div>' +
                           '</div>' +
                           '<button id="close-all-cust-modal" style="background:transparent;border:none;color:#94a3b8;font-size:24px;cursor:pointer;padding:4px 12px;border-radius:8px;">✕</button>';

        // Toolbar
        var toolbar = document.createElement("div");
        toolbar.style = "padding:14px 24px;background:#1e293b;border-bottom:1px solid #334155;display:flex;justify-content:space-between;align-items:center;gap:12px;";
        toolbar.innerHTML = '<div style="display:flex;gap:10px;flex:1;align-items:center;">' +
                            '<input type="text" id="cust-search-input" placeholder="🔍 Tìm Mã KH, Tên KH, SĐT, VinClub User ID..." style="flex:1;max-width:450px;background:#0f172a;border:1px solid #475569;border-radius:8px;padding:8px 14px;color:#ffffff;font-size:14px;outline:none;">' +
                            '<select id="cust-filter-select" style="background:#0f172a;border:1px solid #475569;border-radius:8px;padding:8px 12px;color:#ffffff;font-size:13px;outline:none;">' +
                            '<option value="all">Tất cả Khách hàng (' + totalCount + ')</option>' +
                            '<option value="vc">Chỉ KH có VinClub (' + vcCount + ')</option>' +
                            '<option value="novc">KH chưa có VinClub (' + (totalCount - vcCount) + ')</option>' +
                            '</select>' +
                            '</div>' +
                            '<div style="display:flex;gap:10px;">' +
                            '<button id="btn-copy-all-csv" style="background:#2563eb;color:white;border:none;padding:8px 16px;border-radius:8px;font-weight:600;cursor:pointer;display:flex;align-items:center;gap:6px;">📋 Copy CSV Toàn Bộ</button>' +
                            '</div>';

        // Table Body
        var contentDiv = document.createElement("div");
        contentDiv.style = "flex:1;overflow-y:auto;padding:0;";
        
        function renderTable() {
            var filterText = (document.getElementById("cust-search-input") ? document.getElementById("cust-search-input").value : "").toLowerCase().trim();
            var filterType = document.getElementById("cust-filter-select") ? document.getElementById("cust-filter-select").value : "all";

            var filtered = list.filter(function(item) {
                if (filterType === "vc" && !item.co_vinclub) return false;
                if (filterType === "novc" && item.co_vinclub) return false;

                if (!filterText) return true;
                return item.ma_kh.toLowerCase().indexOf(filterText) !== -1 ||
                       item.ten_kh.toLowerCase().indexOf(filterText) !== -1 ||
                       item.sdt.toLowerCase().indexOf(filterText) !== -1 ||
                       item.vclub_user_id.toLowerCase().indexOf(filterText) !== -1 ||
                       item.one_id.toLowerCase().indexOf(filterText) !== -1;
            });

            var html = '<table style="width:100%;border-collapse:collapse;font-size:13px;text-align:left;">' +
                       '<thead style="position:sticky;top:0;background:#0f172a;color:#94a3b8;border-bottom:1px solid #334155;z-index:2;">' +
                       '<tr>' +
                       '<th style="padding:12px 16px;">STT</th>' +
                       '<th style="padding:12px 16px;">Mã KH DMS</th>' +
                       '<th style="padding:12px 16px;">Tên Khách Hàng</th>' +
                       '<th style="padding:12px 16px;">Số Điện Thoại</th>' +
                       '<th style="padding:12px 16px;">VinClub User ID</th>' +
                       '<th style="padding:12px 16px;">Hạng VinClub</th>' +
                       '<th style="padding:12px 16px;">Mã OneID</th>' +
                       '<th style="padding:12px 16px;">Đơn VSO Liên Quan</th>' +
                       '</tr>' +
                       '</thead><tbody>';

            if (filtered.length === 0) {
                html += '<tr><td colspan="8" style="padding:32px;text-align:center;color:#64748b;">Không tìm thấy kết quả nào.</td></tr>';
            } else {
                filtered.forEach(function(item, idx) {
                    var bg = idx % 2 === 0 ? "rgba(255,255,255,0.02)" : "transparent";
                    var vcBadge = item.co_vinclub ? 
                        '<span style="background:rgba(16,185,129,0.15);color:#34d399;padding:3px 8px;border-radius:6px;font-weight:600;border:1px solid rgba(16,185,129,0.3);">' + (item.hang_vinclub || "Member") + '</span>' : 
                        '<span style="color:#64748b;">Chưa áp dụng</span>';

                    html += '<tr style="background:' + bg + ';border-bottom:1px solid rgba(255,255,255,0.05);">' +
                            '<td style="padding:12px 16px;color:#64748b;">' + (idx + 1) + '</td>' +
                            '<td style="padding:12px 16px;font-weight:600;color:#38bdf8;">' + item.ma_kh + '</td>' +
                            '<td style="padding:12px 16px;font-weight:600;color:#ffffff;">' + item.ten_kh + '</td>' +
                            '<td style="padding:12px 16px;color:#cbd5e1;">' + (item.sdt || '<span style="color:#64748b;">---</span>') + '</td>' +
                            '<td style="padding:12px 16px;font-family:monospace;color:#f59e0b;">' + (item.vclub_user_id || '<span style="color:#64748b;">---</span>') + '</td>' +
                            '<td style="padding:12px 16px;">' + vcBadge + '</td>' +
                            '<td style="padding:12px 16px;color:#cbd5e1;">' + (item.one_id || '<span style="color:#64748b;">---</span>') + '</td>' +
                            '<td style="padding:12px 16px;color:#94a3b8;font-size:12px;">' + (item.cac_don_vso.join(", ") || "---") + '</td>' +
                            '</tr>';
                });
            }

            html += '</tbody>mtable></table>';
            contentDiv.innerHTML = html;
        }

        modal.appendChild(header);
        modal.appendChild(toolbar);
        modal.appendChild(contentDiv);
        overlay.appendChild(modal);
        document.body.appendChild(overlay);

        renderTable();

        document.getElementById("close-all-cust-modal").onclick = function() { overlay.remove(); };
        document.getElementById("cust-search-input").oninput = renderTable;
        document.getElementById("cust-filter-select").onchange = renderTable;

        document.getElementById("btn-copy-all-csv").onclick = function() {
            var csv = "Mã KH DMS,Tên Khách Hàng,Số Điện Thoại,VinClub User ID,Hạng VinClub,Mã OneID Vingroup,Đơn Hàng VSO\n";
            list.forEach(function(item) {
                csv += '"' + item.ma_kh + '","' + item.ten_kh + '","' + item.sdt + '","' + item.vclub_user_id + '","' + (item.hang_vinclub || (item.co_vinclub ? "Member" : "")) + '","' + item.one_id + '","' + item.cac_don_vso.join("; ") + '"\n';
            });
            navigator.clipboard.writeText(csv).then(function() {
                alert("📋 Đã sao chép danh sách TOÀN BỘ " + list.length + " Mã KH định dạng CSV thành công!");
            });
        };
    }
})();
